# Evaluate Boolean Binary Tree. A case is the circuit in level order (null for
# a missing child): 0 and 1 are wires, 2 is OR, 3 is AND; the result is its output.

@example The inner AND is true, so the OR is too, and the root AND sees two trues.
[3,2,1,3,0,null,null,1,1]
=> true

@example The AND has a false input and the lower OR has no true one: false.
[2,3,2,0,1,0,0]
=> false

# A single wire carrying false.
[0]
=> false

# A single wire carrying true.
[1]
=> true

# OR of false and true: swapping the operators would say false.
[2,0,1]
=> true

# AND of true and false: swapping the operators would say true.
[3,1,0]
=> false

# Gate values are not truthy: the OR below the root outputs false.
[3,2,3,0,0,1,1]
=> false

# Mixed levels, where swapping every operator flips the answer.
[2,3,3,1,0,0,1]
=> false

# Two ORs under an AND, each with one true wire.
[3,2,2,1,0,0,1]
=> true

# An OR caterpillar decided by its deepest wire.
[2,0,2,null,null,0,2,null,null,0,1]
=> true

# An AND caterpillar decided by its deepest wire.
[3,1,3,null,null,1,3,null,null,1,0]
=> false

# A left-leaning OR caterpillar.
[2,2,0,2,0,null,null,0,1]
=> true

# An OR whose true input comes from a chain of AND gates.
[2,0,3,null,null,1,3,null,null,1,1]
=> true
