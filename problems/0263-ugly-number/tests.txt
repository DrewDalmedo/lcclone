# Ugly Number.

@example 360 = 2^3 * 3^2 * 5.
360
=> true

@example 66 = 2 * 3 * 11, and 11 is a prime larger than 5.
66
=> false

@example A negative number is never ugly, even though 30 = 2 * 3 * 5 is.
-30
=> false

# 1 = 2^0 * 3^0 * 5^0 is ugly.
1
=> true

# Zero is not positive (and stays divisible by 2 however often it is
# divided).
0
=> false

# The smallest prime larger than 5.
7
=> false

# A repeated factor: 50 = 2 * 5^2, so 5 must be divided out twice.
50
=> true
