# Nim Game.

@example Take 2 and leave 4: whatever your opponent then takes, you take the rest.
6
=> true

@example Any first move leaves 5, 6 or 7; your opponent then leaves you 4, and whatever you take from 4, they take the rest.
8
=> false

@example Take all three pebbles at once.
3
=> true

# The smallest piles: one move takes them whole.
1
=> true

2
=> true

# The smallest pile you lose: too many to take at once, and whatever you
# take, your opponent takes the rest.
4
=> false

# One more than a multiple of 4: take a single pebble.
13
=> true

# Mid-size piles. Working out every smaller pile by recursion goes about n
# calls deep.
1000000
=> false

100000002
=> true

# 2^30, a large power of two.

=> false

2147483644
=> false

=> true
