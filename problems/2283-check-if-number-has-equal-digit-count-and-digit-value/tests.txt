# Check if Number Has Equal Digit Count and Digit Value.

@example Position 0 claims two 0s, position 1 one 1, position 2 two 2s, and positions 3 and 4 no 3s and no 4s: all correct.
"21200"
=> true

@example Position 0 claims one 0, which is right, but position 1 claims two 1s and there is only one.
"120"
=> false

@example Position 0 claims there is no 0, yet that very character is a 0.
"0"
=> false

# The other true strings: of length 4, 7, 8, 9 and 10.
"2020"
=> true

"3211000"
=> true

"42101000"
=> true

"521001000"
=> true

"6210001000"
=> true

# A single digit other than 0 claims that many 0s, and there are none.
"1"
=> false

# All zeros: position 0 claims no 0s.
"0000000000"
=> false

# Each digit d occurs d times, but that is not what is asked.
"122"
=> false

# Positions 0 to 8 are right; only position 9 is wrong (it claims no 9s).
"9000000000"
=> false

# A digit larger than n - 1 claims too many of its position's digit.
"21209"
=> false

# The digits of a true string, reversed: they still add up to n.
"00212"
=> false

# A true string with one digit changed, removed or added.
"3210000"
=> false

"621000100"
=> false

"212000"
=> false
