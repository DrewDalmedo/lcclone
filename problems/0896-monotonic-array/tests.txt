# Monotonic Array.

@example No step drops. The level steps from 3 to 3 and from 10 to 10 are allowed.
[3,3,8,10,10,15]
=> true

@example No step climbs.
[20,-4,-4,-9]
=> true

@example The step from 6 to 1 drops, and the step from 1 back to 6 climbs.
[6,6,1,6]
=> false

# One element: no steps at all.
[-7]
=> true

# Two elements always qualify, here the two extremes.
[-100000,100000]
=> true

# All equal: only level steps.
[5,5,5,5]
=> true

# Level steps first, then a drop: the first step doesn't tell the direction.
[9,9,9,4]
=> true

# Level steps first, then a climb and a drop.
[0,0,0,1,0]
=> false

# The only drop is the last step.
[1,2,3,4,5,4]
=> false

# The only climb is the first step.
[8,9,7,4,4,1]
=> false

# A drop and a climb far apart, with level steps between them.
[3,1,1,1,1,1,1,5]
=> false

# Up and back down to the starting value: the first and last values match.
[4,6,8,4]
=> false

# Steps of 200000 up and 150000 down: their product, -3 * 10^10, wraps to a
# positive number in a 32-bit int.
[-100000,100000,-50000]
=> false

# Steps of 100000 and 30000 up: their product, 3 * 10^9, wraps to a
# negative number in a 32-bit int.
[-100000,0,30000]
=> true

# From the largest value to the smallest, with level steps, one at the end.
[100000,100000,0,0,-100000,-100000]
=> true

@gen 1 small
[0,0,0,0,-3,-3,-3]
=> true

@gen 2 small
[-3,-7,0,1,3]
=> false

@gen 3 small
[100000,99997,100000]
=> false

=> true
