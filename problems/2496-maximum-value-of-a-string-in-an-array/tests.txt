# Maximum Value of a String in an Array.

@example "07" is read as 7, which beats "road" (4 characters), "k2" (2) and "x" (1).
["k2","road","07","x"]
=> 7

@example "00000003" has 8 characters but is worth only 3; "boat" is worth 4.
["00000003","q5","boat"]
=> 4

@example Codes made only of zeros are worth 0.
["0","00"]
=> 0

# The smallest inputs: one letter, one digit, one zero.
["a"]
=> 1

["5"]
=> 5

["0"]
=> 0

# Several all-zero codes, the longest nine characters long.
["000000000","0","00","0000"]
=> 0

# The largest number wins over the longest letter code.
["abcdefghi","999999999","1"]
=> 999999999

# Letters at the end, at the start and in the middle make a code worth its
# length, not the number its digits begin with.
["98765432x","8"]
=> 9

["a99999999","5"]
=> 9

["9999a9999","10"]
=> 10

["12345678z","3","40"]
=> 40

# A long zero-padded code is worth its small number, not its length.
["000000001","ab","3"]
=> 3

["000000007","00000099"]
=> 99

# A number equal to a letter code's length.
["4","abcd"]
=> 4

# Letters only.
["xyz","pq","longwords"]
=> 9

# Numbers past what a byte or a 16-bit integer can hold.
["256","b","300"]
=> 300

["65537","300","b"]
=> 65537

# Many codes; the winner, a number, is the last of them.
["a1","b22","c333","d4444","e55555","f666666","g7777777","h88888888","012345678"]
=> 12345678
