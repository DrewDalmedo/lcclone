# Coupon Code Validator.

@example "snack 5" has a space and "Pharmacy" a capital; the other three come in the order of their lines of business.
["MEDS4U","snack_5","Tv2024","snack 5","AAA"]
["pharmacy","grocery","electronics","grocery","Pharmacy"]
[true,true,true,true,true]
=> ["Tv2024","snack_5","MEDS4U"]

@example One line of business: digits sort before capitals, capitals before _, and _ before small letters. "b" is switched off.
["b2","B2","_b","2b","b"]
["restaurant","restaurant","restaurant","restaurant","restaurant"]
[true,true,true,true,false]
=> ["2b","B2","_b","b2"]

@example The first code is empty, and "groceries" is not a line of business.
["","OK1"]
["grocery","groceries"]
[true,true]
=> []

# A single coupon that passes.
["X"]
["electronics"]
[true]
=> ["X"]

# A single coupon that is switched off.
["X"]
["electronics"]
[false]
=> []

# Codes made only of underscores pass; a prefix sorts first.
["a_b","___","_"]
["grocery","grocery","grocery"]
[true,true,true]
=> ["_","___","a_b"]

# One bad character, at the end, in the middle, at the start, or a lone space.
["CODE ","a.b","x-y","50%","ok@","$1"," ","fine"]
["pharmacy","pharmacy","pharmacy","pharmacy","pharmacy","pharmacy","pharmacy","pharmacy"]
[true,true,true,true,true,true,true,true]
=> ["fine"]

["grocery","grocery","grocery","grocery"]

["Z9","A1","Z9"]
["restaurant","restaurant","electronics"]
[true,true,true]
=> ["Z9","A1","Z9"]
