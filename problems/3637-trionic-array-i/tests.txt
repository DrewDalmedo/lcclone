# Trionic Array I.

@example Up from -4 to 7, down to 2, then up to 5.
[-4,0,7,2,5]
=> true

@example The two 8s side by side make a horizontal segment.
[3,8,8,1,6]
=> false

@example Up, down, up, and then down again: one part too many.
[6,9,2,7,1]
=> false

# Three values leave no room for three parts.
[-1000,1000,-1000]
=> false

# The shortest trionic array: each part is a single step.
[1,4,2,3]
=> true

# Only rises.
[-3,-2,0,5,9]
=> false

# Starts by going down.
[9,4,6,1,8]
=> false

# Rises, then falls to the end.
[1,5,9,6,2]
=> false

# Falls, then rises.
[7,3,0,4,8]
=> false

# Falls, rises, then falls: it turns exactly twice, but the parts come in
# the wrong order.
[4,-2,-7,5,9,1]
=> false

# Five parts.
[0,2,1,3,2,4]
=> false

# A horizontal segment inside the last part, and one at the bottom of the
# fall.
[1,4,2,3,3]
=> false

[5,9,2,2,6]
=> false

# The turning points are not the largest and smallest values: the end
# climbs above the peak, and the start lies below the valley.
[-1000,999,-999,1000]
=> true

# A long falling part between two single steps.
[0,90,80,70,60,50,40,30,95]
=> true

# A long last part.
[10,20,-5,-4,-3,-2,-1,0,1]
=> true
