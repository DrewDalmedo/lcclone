# Furthest Point From Origin.

@example Filling both blanks with L gives "LLLLR": four steps left and one right finish at -3.
"_LL_R"
=> 3

@example There are no blanks. The token is 4 to the right after four steps, but it finishes at -2.
"RRRRLLLLLL"
=> 2

@example The two forced steps cancel out, and the three blanks, filled in alike, carry the token 3 from 0.
"R__L_"
=> 3

# One step of each kind.
"L"
=> 1

"R"
=> 1

"_"
=> 1

# Left leads, and the blanks must go left too: sending them right would
# finish at 1.
"LL_LR_L"
=> 5

# Right leads, and the blanks must go right: sending them left would finish
# at 0.
"_RR_LR"
=> 4

# Blanks at the start, and at the end.
"__RRR"
=> 5

"LLL__"
=> 5

# The token strays 8 to the right before it comes back past 0.
"RRRRRRRRLLLLLLLLLL_"
=> 3

# The longest scripts: all blanks; as many L as R; all L.
"__________________________________________________"
=> 50

"LRLRLRLRLRLRLRLRLRLRLRLRLRLRLRLRLRLRLRLRLRLRLRLRLR"
=> 0

"LLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLLL"
=> 50

# 50 steps: 25 L, 12 R and 13 blanks.
"LLR_LLR_LLR_LLR_LLR_LLR_LLR_LLR_LLR_LLR_LLR_LLR_L_"
=> 26
