# Find First Palindromic String in the Array.

@example In moon the outer letters m and n differ; noon mirrors itself and comes before level, which mirrors itself too.
["moon","noon","level"]
=> "noon"

@example A one-letter word is always a palindrome.
["ox","q","eve"]
=> "q"

@example No word mirrors itself.
["top","pot","stop"]
=> ""

# A single one-letter word.
["k"]
=> "k"

# A single word of two different letters.
["ab"]
=> ""

# Two equal letters.
["zz"]
=> "zz"

# Only the outermost pair differs: a check that starts at the second
# letter would accept "botatoq".
["botatoq","pop"]
=> "pop"

# Only the innermost pair of an even-length word differs.
["xyzwyx","gig"]
=> "gig"

# A half followed by the same half, not mirrored: only "tattat", whose
# half is itself a palindrome, mirrors.
["abcabc","tattat"]
=> "tattat"

# Equal first and last letters are not enough.
["rover","refer"]
=> "refer"

# The first palindrome is shorter than a later one...
["dad","rotator"]
=> "dad"

# ... or longer than a later one.
["deified","bob"]
=> "deified"

# Repeated words.
["no","on","no","xyx","xyx"]
=> "xyx"
