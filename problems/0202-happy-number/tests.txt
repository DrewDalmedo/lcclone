# Happy Number.

@example 239 -> 94 -> 97 -> 130 -> 10 -> 1, so 239 is happy.
239
=> true

@example 15 -> 26 -> 40 -> 16 -> 37 -> 58 -> 89 -> 145 -> 42 -> 20 -> 4 -> 16: 16 comes round again, and 1 is never reached.
15
=> false

@example 7 -> 49 -> 97 -> 130 -> 10 -> 1: a single digit other than 1 can still be happy.
7
=> true

# 1 is happy without taking a step.
1
=> true

# 4 lies on the loop that every unhappy number ends in.
4
=> false

# Another single digit; only 1 and 7 are happy.
9
=> false

# One step to 1: zeros add nothing.
10
=> true

1000000000
=> true

# Through 7 on the way: 2111 -> 7 -> 49 -> ... (6 steps).
2111
=> true

# 7 steps, the most any n in range needs:
# 78999 -> 356 -> 70 -> 49 -> 97 -> 130 -> 10 -> 1.
78999
=> true

# Ten digits whose squares also add up to 356.
1099995211
=> true

# An unhappy number that visits 19 different values before one repeats.
269
=> false

2147483645
