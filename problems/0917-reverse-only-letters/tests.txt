# Reverse Only Letters.

@example The letters G, o, i, t come back as t, i, o, G on the same four indices; the uppercase G moves to index 4, and 4 and ! stay put.
"Go4it!"
=> "ti4oG!"

@example [, ], _ and ^ are not letters, so only a, b, C, d trade places.
"[ab]_C^d"
=> "[dC]_b^a"

@example There is no letter at all, so nothing moves.
"2+2=4"
=> "2+2=4"

# A single letter, alone and between non-letters.
"q"
=> "q"

"1&x&2"
=> "1&x&2"

# A single non-letter.
"#"
=> "#"

# Only letters: the whole string reverses, each letter keeping its case.
"abcDE"
=> "EDcba"

# No letters, but the characters between Z and a, and @ just before A.
"[]^_`@"
=> "[]^_`@"

# Letters next to @ [ ` ] ^ _.
"@Z[`a]^_"
=> "@a[`Z]^_"

# Digits between the letters.
"a1b2c3d"
=> "d1c2b3a"

# Long runs of non-letters on both sides of the letters.
"!!!??abc??!!!"
=> "!!!??cba??!!!"

# Letters only at the two ends.
"M%%%%%%n"
=> "n%%%%%%M"

# An odd number of letters: the middle one stays where it is.
"pq.r.st"
=> "ts.r.qp"

# Alternating, starting and ending with a non-letter.
".a.B.c.D."
=> ".D.c.B.a."

# Repeated letters.
"aab"
=> "baa"

"x=x=y"
=> "y=x=x"

# The longest string: 50 letters, then 50 digits.
"abcdefghijklmnopqrstuvwxyABCDEFGHIJKLMNOPQRSTUVWXY01234567890123456789012345678901234567890123456789"
=> "YXWVUTSRQPONMLKJIHGFEDCBAyxwvutsrqponmlkjihgfedcba01234567890123456789012345678901234567890123456789"
