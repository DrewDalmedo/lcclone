# Split Strings by Separator. The chunks are listed string by string, left
# to right within each string; [] when there are none.

@example Each # marks a boundary; nothing lies between the two # after blue, and violet has no # at all.
["red#green","#blue##","violet"]
"#"
=> ["red","green","blue","violet"]

@example Only | marks boundaries: the commas are ordinary characters, and one of them is a chunk by itself.
["a,b|c","d|,|e"]
"|"
=> ["a,b","c","d",",","e"]

@example No @ stands between ab and cd, but they come from different strings, so they stay separate chunks.
["ab","cd@","@ef"]
"@"
=> ["ab","cd","ef"]

# A single string of one letter.
["q"]
"."
=> ["q"]

# A lone separator has no chunk.
["."]
"."
=> []

# Every string is separators only.
["$$$$","$","$$"]
"$"
=> []

# Every mark in one string: only the separator cuts it.
["a.b,c|d$e#f@g"]
"."
=> ["a","b,c|d$e#f@g"]

["a.b,c|d$e#f@g"]
"$"
=> ["a.b,c|d","e#f@g"]

# A chunk made only of other punctuation marks.
["x#.,|$@#y"]
"#"
=> ["x",".,|$@","y"]

# Runs of separators at the start, in the middle and at the end.
[",,ab,,,,cd,,"]
","
=> ["ab","cd"]

# 20 characters and no separator: each string is one chunk, letters or marks.
["abcdefghijklmnopqrst","@#$|,@#$|,@#$|,@#$|,"]
"."
=> ["abcdefghijklmnopqrst","@#$|,@#$|,@#$|,@#$|,"]

# Ten chunks from a string of 20 characters, the most there can be.
["|a|b|c|d|e|f|g|h|i|j","a|b|c|d|e|f|g|h|i|j|"]
"|"
=> ["a","b","c","d","e","f","g","h","i","j","a","b","c","d","e","f","g","h","i","j"]

# Equal chunks are all listed, in order.
["go.go","go","..go."]
"."
=> ["go","go","go","go"]

# The separator appears in one string only.
["a$b","c.d"]
"$"
=> ["a","b","c.d"]

# The strings keep their order; nothing is sorted.
["zeta|alpha","mid"]
"|"
=> ["zeta","alpha","mid"]

# Random small cases from gen.c (lc check --regen makes them).
@gen 1 small
["@#|a||","||||"]
"|"
=> ["@#","a"]

@gen 2 small
["##|"]
"#"
=> ["|"]

@gen 4 small
[",",",a,|a","a|,"]
","
=> ["a","|a","a|"]

@gen 5 small
["#$$$","b$"]
"$"
=> ["#","b"]

@gen 6 small
["$$a","$","$b@$","$","b#ba"]
"$"
=> ["a","b@","b#ba"]

@gen 7 small
["$@$a$","$$$","$a$$$b","ba","$$$"]
"$"
=> ["@","a","a","b","ba"]

@gen 10 small
[",,,,,","bab,|$",",,,,|"]
","
=> ["bab","|$","|"]
