# Base 7.

@example 52 = 1 * 49 + 0 * 7 + 3: a one in the 49s place, no sevens, and three ones.
52
=> "103"

@example 20 = 2 * 7 + 6, and the minus sign goes in front.
-20
=> "-26"

@example Zero has no nonzero digit, so it is written as a single 0.
0
=> "0"

# The largest digit, then the first number with two digits.
6
=> "6"

7
=> "10"

# The smallest negative magnitude.
-1
=> "-1"

# A negative number ending in zeros: 49 = 7^2.
-49
=> "-100"

# 7^2 - 1, every digit a 6.
48
=> "66"

# Zeros inside and at the end of the digits: 7^4 + 1, and 3 * 7^4.
2402
=> "10001"

7203
=> "30000"

# 7^8 - 1, the largest number with 8 digits, and 7^8, the smallest with 9.
5764800
=> "66666666"

5764801
=> "100000000"

# The top of the range.
10000000
=> "150666343"

# The bottom of the range, and -7^8: ten characters with the sign, so a
# buffer of 10 bytes has no room for the NUL.
-10000000
=> "-150666343"

-5764801
=> "-100000000"
