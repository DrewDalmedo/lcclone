# Find if Digit Game Can Be Won.

@example The short cards add up to 6 + 8 + 2 + 7 = 23. Claiming the long card scores 31 against 23.
[6,31,8,2,7]
=> true

@example The long cards add up to 13 + 19 = 32 and the short ones to 8 + 9 + 7 + 8 = 32, so either claim ties.
[13,8,9,19,7,8]
=> false

@example There are no short cards. Claiming the long ones takes everything, 59 against 0.
[12,47]
=> true

# One card, the smallest number.
[1]
=> true

# One card, the largest number.
[99]
=> true

# Only short cards; equal numbers are separate cards.
[4,4,4]
=> true

# 9 is short and 10 is long: 10 against 9 + 1 ties. Counting 9 as long,
# or 10 as short, would break the tie.
[10,9,1]
=> false

# 99 is long: it ties with eleven 9s.
[99,9,9,9,9,9,9,9,9,9,9,9]
=> false

# A tie with many cards on each side: 11 + 12 + 13 + 14 + 15 = 65 against
# twelve short cards that also add up to 65.
[11,9,12,8,13,7,14,6,15,5,4,3,2,1,9,8,3]
=> false

# More short cards than long ones, but equal totals.
[10,5,5]
=> false

# As many short cards as long ones, but the long ones add up to more.
[10,20,9,1]
=> true

# The short cards win, 7 + 9 + 8 + 9 = 33 against 30.
[30,7,9,8,9]
=> true
