# Merge Two Binary Trees. A case is root1, then root2, each in level order;
# the result is the combined tree in level order.

@example 4 + 1 and -2 + 5 are sums; 3 and 9 come only from root2, and 7 only from root1.
[4,-2,null,7]
[1,5,3,null,9]
=> [5,3,3,7,9]

@example The roots cancel to 0, which stays, with 6 and its child -1 from root2 and 3 from root1 below it.
[-8,null,3]
[8,6,null,null,-1]
=> [0,6,3,null,-1]

@example With nothing in root1, the result is root2 as it stands.
[]
[2,null,6]
=> [2,null,6]

# Both empty.
[]
[]
=> []

# Only the first tree has nodes.
[9,null,2]
[]
=> [9,null,2]

# One node each, at the lowest values.
[-10000]
[-10000]
=> [-20000]

# One node each, at the highest values.
[10000]
[10000]
=> [20000]

# Every sum is 0, and every node stays.
[5,3,-2]
[-5,-3,2]
=> [0,0,0]

# A deep subtree of root1 under a spot root2 lacks.
[6,2,null,3,null,4]
[-4]
=> [2,2,null,3,null,4]

# A deep subtree of root2 under a spot root1 lacks.
[7]
[3,null,2,null,3,null,4]
=> [10,null,2,null,3,null,4]

# Shapes that share only the root.
[1,2,null,3]
[1,null,4,null,5]
=> [2,2,4,3,null,null,5]

# The same shape: every value is a sum.
[1,2,3,4,5]
[10,20,30,40,50]
=> [11,22,33,44,55]

# Left and right must not cross.
[1,2,3,null,4]
[5,null,6,7]
=> [6,2,9,null,4,7]

# Each tree carries a two-level subtree on its own side.
[3,null,2,null,8]
[4,5,null,null,7,6]
=> [7,5,2,null,7,null,8,6]

# Small random trees, with ties and one-sided shapes.
@gen 1 small
[]
[-3]
=> [-3]

@gen 2 small
[-3,0,-3,null,-1]
[0,1,-3,null,0,3,null,null,-3]
=> [-3,1,-6,null,-1,3,null,null,-3]

@gen 3 small
[-1,3,1,null,-3,1,null,null,null,null,2]
[-3,null,2,-2]
=> [-4,3,3,null,-3,-1,null,null,null,null,2]

@gen 4 small
[]
[]
=> []

@gen 5 small
[]
[1,null,-3,-1,null,null,0,null,-3,-2]
=> [1,null,-3,-1,null,null,0,null,-3,-2]

@gen 6 small
[3,0,null,-2,null,-1,-1]
[2,-1,-1,null,null,-1,0,null,-3]
=> [5,-1,-1,-2,null,-1,0,-1,-1,null,-3]
