# Contains Duplicate II.

@example The 8s at positions 1 and 3 are 2 apart, and k = 2.
[3,8,6,8]
2
=> true

@example The only repeat, the 4s, is 3 apart, more than k = 2.
[4,0,7,4]
2
=> false

@example The first two 9s are 4 apart, too far, but the last two are 2 apart.
[9,1,2,3,9,0,9]
2
=> true

# One element: nothing to pair with, however large k is.
[5]
100000
=> false

# k = 0 is always false, even for equal neighbours.
[7,7]
0
=> false

# Equal neighbours with k = 1.
[7,7]
1
=> true

# A repeat at distance exactly k counts; at distance k + 1 it doesn't.
[12,3,7,-1,5,12]
5
=> true

[12,3,7,-1,5,12]
4
=> false

# k reaches past the end of the array: any repeat counts.
[6,-2,11,0,6]
100000
=> true

# A value four times: only its last two positions are close enough, so
# remembering where a value was first seen is not enough.
[-1,4,6,8,-1,3,5,7,-1,-1]
1
=> true

# The 3s are too far apart; the near repeat is the 2s, later on.
[3,5,9,3,2,2]
2
=> true

# Two repeats, each one position too far apart.
[10,20,30,40,10,20]
3
=> false

# Both bounds.
[1000000000,-1000000000,1000000000]
1
=> false

[-1000000000,1000000000,0,-1000000000]
3
=> true

# 4 and -4 are different values; the two 4s are 2 apart.
[4,-4,4]
1
=> false

# No repeat, a 0 at the end and k past the end: a scan that looks one
# position beyond the last must not find a 0 there.
[-5,3,0]
4
=> false

# Different values that agree in their low 20 bits and modulo 10^6: a
# table indexed by part of the value must still compare the values.
[0,1048576,-1048576,3000000]
3
=> false

4
=> true

=> false

[2,-1,-1,0,2,-1,1]
