# Replace All Digits with Characters.

@example The 0 repeats m, the 3 counts on from o to r, and the 1 from w to x.
"m0o3w1"
=> "mmorwx"

@example The last character is a letter, so it stays as it is.
"c4t2z"
=> "cgtvz"

@example Nine places on from q is z, the farthest a digit can reach.
"q9"
=> "qz"

# A single letter: nothing to replace.
"a"
=> "a"

# After z only a 0 is possible.
"z0"
=> "zz"

# Each digit counts on from its own letter, not from what the digit
# before it became.
"a2a2a2a"
=> "acacaca"

# Different letters and digits that all land on j.
"a9b8c7d6e5"
=> "ajbjcjdjej"

# A digit counts on from the letter before it, not the one after it.
"x1a1"
=> "xyab"

# Every digit 0: each letter of the alphabet doubled.
"a0b0c0d0e0f0g0h0i0j0k0l0m0n0o0p0q0r0s0t0u0v0w0x0y0z0"
=> "aabbccddeeffgghhiijjkkllmmnnooppqqrrssttuuvvwwxxyyzz"

# Every digit from 0 to 9 after the same letter.
"f0f1f2f3f4f5f6f7f8f9"
=> "fffgfhfifjfkflfmfnfo"

# The longest string, 100 characters, ending in a digit.
"p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9p9"
=> "pypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypypy"

# 99 characters, ending in a letter.
"y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y1y"
=> "yzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzyzy"
