# Count Asterisks.

@example Indoors are "**" and "*c*"; one star is before the first bar, one between the 2nd and 3rd bars and one after the last bar.
"*a|**|b*|*c*|*"
=> 3

@example With no bars, every star is outdoors.
"q**w*e"
=> 3

@example All three stars lie between the 1st and 2nd bars.
"|*k**|"
=> 0

# One character.
"*"
=> 1

"m"
=> 0

# Letters only.
"zigzag"
=> 0

# Bars only.
"||"
=> 0

"||||||"
=> 0

# Two bars side by side enclose nothing, so both stars are outdoors
# (splitting with strtok loses the empty stretch and counts 1).
"*||*"
=> 2

# A leading bar: the empty start is outdoors, the stretch "a" indoors.
"|a|*"
=> 1

"|**|***"
=> 3

# Stars between a closing bar and the next opening bar count; stars
# between the first and last bar are not all indoors.
"|*|**|*|"
=> 2

"k*|x|**|*|p*"
=> 4

# Many tiny regions: the same four stars, all outdoors, then all indoors.
"*||*||*||*"
=> 4

"|*||*||*||*|"
=> 0

# 1000 stars and no bars: the largest answer, past what a char counter holds.
"****************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************"
=> 1000

# One region spanning the whole string.
"|**************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************************|"
=> 0

# 256 stars outdoors around an indoor stretch: a byte-sized counter wraps to 0.
"**********************************************************************************************************************************|*a*|******************************************************************************************************************************"
=> 256

# Random small cases from gen.c (lc check --regen makes them).
@gen 1 small
