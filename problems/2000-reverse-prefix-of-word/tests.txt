# Reverse Prefix of Word.

@example The first "n" ends the head "plan", which becomes "nalp"; the tail "et" stays.
"planet"
"n"
=> "nalpet"

@example "a" first turns up at index 1, so only "ba" is turned around; the later "a"s are in the tail.
"banana"
"a"
=> "abnana"

@example There is no "e" in the word, so it comes back unchanged.
"rhythm"
"e"
=> "rhythm"

# One letter, equal to ch or not.
"q"
"q"
=> "q"

"q"
"w"
=> "q"

# ch is the first letter: the head is that letter alone, even though ch
# comes back later.
"seasons"
"s"
=> "seasons"

# ch only at the end: the whole word turns around.
"stream"
"m"
=> "maerts"

# The head ends at ch itself, not one letter before or after it.
"pqrstu"
"r"
=> "rqpstu"

# A second ch right after the first one stays in the tail.
"xyzzy"
"z"
=> "zyxzy"

# The first ch decides, not the last one.
"mississippi"
"s"
=> "simsissippi"

# Every letter but ch.
"zyxwvutsrqponlkjihgfedcba"
"m"
=> "zyxwvutsrqponlkjihgfedcba"

# The same letter throughout, and one different letter at the end.
"aaaa"
"a"
=> "aaaa"

"bbbba"
"a"
=> "abbbb"
