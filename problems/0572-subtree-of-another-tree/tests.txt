# Subtree of Another Tree. A case is root, then subRoot, each a tree in level
# order with null for a missing child.

@example Cutting above 15 drops exactly 15 with its right child 20.
[10,6,15,2,8,null,20]
[15,null,20]
=> true

@example 9 and 1 are there, but the branch at 9 goes on to 4.
[7,3,9,null,5,1,null,null,null,4]
[9,1]
=> false

@example The leaf -1 is a branch of one node.
[12,-1,30,null,null,8]
[-1]
=> true

# One node each, equal.
[5]
[5]
=> true

# One node each, with opposite signs.
[5]
[-5]
=> false

# subRoot is the whole of root.
[2,4,6,null,8]
[2,4,6,null,8]
=> true

# subRoot has more nodes than root.
[1]
[1,1]
=> false

# A single node matches a leaf only, and this 4 has a child.
[1,4,2,3]
[4]
=> false

# The first node holding 5 fails, and the second one matches.
[1,5,5,2,null,6,7]
[5,6,7]
=> true

# The same values with the children swapped.
[8,2,3]
[8,3,2]
=> false

# The same values with the child on the other side.
[40,30]
[40,null,30]
=> false

# Read as text, 7 is inside 17.
[2,17]
[7]
=> false

# Same size and same sum as the branch at 2.
[6,9,2,4,null,1,3]
[2,2,2]
=> false

# Read as text, 3 is inside -3.
[5,-3]
[3]
=> false

# 0 is a real value, here at the right leaf.
[0,-1,0]
[0]
=> true

# With 0 as the marker for a missing child, 5 over a right child 0 would read
# like 5 over a left child 0.
[1,5,null,null,0]
[5,0]
=> false
