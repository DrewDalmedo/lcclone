# Check if Bitwise OR Has Trailing Zeros.

@example 6 | 10 is 0110 | 1010 = 1110 in binary, which ends in a 0.
[3,6,9,10]
=> true

@example Only 8 ends in 0 in binary. A choice needs a second element, and every other one ends in 1, which the OR keeps.
[5,8,11,7]
=> false

@example The two 12s are different elements, and 12 | 12 = 12 is 1100 in binary.
[12,12]
=> true

# Two odd elements.
[1,1]
=> false

# One element ends in 0, but it can't be chosen alone.
[100,1]
=> false

# Two even elements, at the bounds.
[2,100]
=> true

# All odd.
[99,1,3,77,5,51,9]
=> false

# All even.
[10,24,6,88,40]
=> true

# The only two evens are at the two ends.
[4,1,3,5,7,9,11,6]
=> true

# 2 | 6 = 110 has a single trailing zero, which is enough; it need not be a
# power of two either.
[2,9,6]
=> true

# The OR of the whole array is odd, but 4 | 8 = 1100 is not.
[4,3,8]
=> true

# The same even value twice, apart.
[50,7,50]
=> true
