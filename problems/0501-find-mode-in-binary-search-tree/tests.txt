# Find Mode in Binary Search Tree. A case is the tree in level order; the
# result is every value with the highest count, in any order.

@example Three nodes hold 7, and no other value is held by more than two.
[7,3,9,3,7,null,12,null,null,7]
=> [7]

@example -4 and 6 are each held by two nodes, so both are modes.
[2,-4,6,-4,null,6]
=> [-4,6]

@example No value repeats, so every value is a mode.
[15,8,40]
=> [8,15,40]

# The smallest tree: one node.
[-37]
=> [-37]

# The smallest value is 0: a "previous value" starting at 0 would count one 0 too many and lose the tie.
[5,0,5,0]
=> [0,5]

# The smallest value is -1: the same trap for a "previous value" starting at -1.
[3,-1,3,-1]
=> [-1,3]

# The sole mode is the largest value, so the last run must be closed.
[1,null,4,2,4,null,null,null,4]
=> [4]

# A tie that includes the largest value.
[10,4,20,4,7,null,20]
=> [4,20]

# A longer run after shorter ones: the modes so far are dropped.
[2,1,3,1,null,3,3]
=> [3]

# A long run first, then shorter ones.
[5,5,8,5,null,7,9]
=> [5]

# Three tied runs at different places in the order.
[2,0,7,-3,2,5,7,-3]
=> [-3,2,7]

# Copies of the root's value in both subtrees, each a few levels down.
[4,1,9,null,3,6,null,null,4,4]
=> [4]

# Both extreme values are modes.
[0,-100000,100000,-100000,null,null,100000]
=> [-100000,100000]

# Every value equal, on a zigzag.
[7,7,null,null,7,7]
=> [7]

# A left path of distinct values: all are modes.
[50,40,null,30,null,20]
=> [20,30,40,50]

# Small random trees with many repeats.
@gen 1 small
[5,2,null,2]
=> [2]

@gen 2 small
[100000,100000,null,99997,null,99994,null,99991,null,99991,null,99990,null,null,99990]
=> [99990,99991,100000]

@gen 3 small
[3,1,7,null,1,5,null,null,null,null,6]
=> [1]

@gen 4 small
[-3,null,-3]
=> [-3]

@gen 5 small
[1,null,1,null,1,null,2]
=> [1]

@gen 6 small
[1,0,null,-1,null,-1]
=> [-1]

# Full size, 10^4 nodes: a left path of one value, a right path of distinct
# values, a zigzag whose mode is the last run, a random shape with many ties,
# a complete tree whose mode is the first run (300 copies), a left path of two
# tied halves, and a comb with 1000 copies of one value in the middle.
@full 1 15e843cb
=> [100000]

@full 2 1a9d474e

@full 3 9655c3db
=> [100000]

@full 4 e1aae091

@full 5 08a22987
=> [-100000]

@full 6 f6aee376
=> [-1,0]

@full 7 b7425c79
