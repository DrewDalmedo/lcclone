# Check Array Formation Through Concatenation.

@example Taken in the order [6], [40,2], [17,33], the pieces spell out arr.
[6,40,2,17,33]
[[17,33],[6],[40,2]]
=> true

@example arr needs 12 before 3, but the piece [3,12] can't be rearranged.
[12,3,57,9]
[[3,12],[57,9]]
=> false

@example No piece holds 71, and the 72 of the second piece is not in arr.
[30,4,71]
[[30],[4,72]]
=> false

# A single number, the same in both.
[100]
[[100]]
=> true

# A single number, different.
[1]
[[2]]
=> false

# One piece that is arr itself.
[9,1,50,77]
[[9,1,50,77]]
=> true

# One piece that is arr reversed.
[9,1,50,77]
[[77,50,1,9]]
=> false

# Every piece has length 1: true exactly when they hold arr's numbers.
[8,3,64,25]
[[25],[8],[64],[3]]
=> true

[8,3,64,25]
[[25],[8],[65],[3]]
=> false

# A piece whose first number fits but whose second doesn't.
[10,20,30,40]
[[10,30],[20,40]]
=> false

# Only the very last number differs.
[2,4,6,8,11]
[[6,8,10],[2,4]]
=> false

# Every piece starts and ends where arr has the same numbers, but the
# middle numbers of the two pieces are swapped.
[21,8,34,55,3,13]
[[55,8,13],[21,3,34]]
=> false

# The first piece of the input is needed.
[7,15,1]
[[15,1],[7]]
=> true

# A number of arr that starts no piece.
[3,50,61]
[[3],[61,50]]
=> false

# The bounds 1 and 100.
[100,1]
[[1],[100]]
=> true

# Pieces of several lengths, in an order unlike arr's.
[13,90,44,5,61,28,77]
[[77],[44,5,61],[13,90],[28]]
=> true
