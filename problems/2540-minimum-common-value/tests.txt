# Minimum Common Value.

@example Both logs hold 8 and 15; 8 comes first.
[3,8,8,15,21]
[5,8,15,30]
=> 8

@example No timestamp is in both logs.
[4,9,12]
[1,6,10,13]
=> -1

@example The last timestamp of the first log is the first one of the second.
[2,5,11]
[11,40]
=> 11

# The smallest input, with a match.
[1]
[1]
=> 1

# The smallest input, without one.
[1]
[2]
=> -1

# Every timestamp of nums1 comes before every timestamp of nums2.
[3,5,8]
[9,12,20]
=> -1

# Every timestamp of nums1 comes after every timestamp of nums2.
[100,200]
[10,20,30]
=> -1

# Interleaved, but never equal.
[1,3,5,7,9]
[2,4,6,8,10]
=> -1

# The only common value is the last of both logs, and the largest allowed.
[1,4,6,1000000000]
[2,3,5,1000000000]
=> 1000000000

# The first timestamp of nums1 is the last of nums2.
[50,60,70]
[10,20,50]
=> 50

# Repeats on both sides; stepping past the larger value misses the match.
[1,1,1,4,4,7]
[2,2,4,4,9]
=> 4

# Two identical logs.
[6,6,9,9,12]
[6,6,9,9,12]
=> 6

# Several common values: the smallest one wins, not the last one met.
[2,4,6,8,10,12]
[3,6,9,12]
=> 6

# A single timestamp against a longer log, matching its last entry.
[13]
[1,3,5,7,9,11,13]
=> 13

# A run of one value in each log.
[7,7,7,7,8]
[3,3,7,7]
=> 7

# Unequal values that agree modulo common table sizes: 655360000 is
# 2^20 * 5^4 (every power of two up to 2^20) and 500000000 is 2^8 * 5^9
# (every power of ten up to 10^8). A hash table that keys on the value
# modulo its size and never compares the values themselves finds a match
# that isn't there.
[7,900]
[500000900,655360007]
=> -1

[50,500000004,655360004]
[4,50]
=> 50

# Random small cases from gen.c (lc check --regen makes them).
@gen 1 small
[3,3,5,8,8,9,9]
[3,3,3,4,7,8,8]
=> 3

[1,5,6]
=> 1

[5,7,7,8]
[3,5,6]

[5,6]
[1,1,1,1,2,2]

[3,12]
[19,22]
=> -1

[2,3,6,6,8,8]
[2,3,4,5,6,9,9]
