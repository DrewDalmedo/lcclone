# Convert a Number to Hexadecimal.

@example 3054 = 11 * 256 + 14 * 16 + 14, and the digits 11, 14 and 14 are written b, e and e.
3054
=> "bee"

@example -256 + 2^32 = 4294967040, which is ffffff00 in base 16.
-256
=> "ffffff00"

@example 4096 = 16^3: a 1 and three zeros, all of which stay.
4096
=> "1000"

# Zero is the only answer that starts with 0.
0
=> "0"

# One digit: the first and the last letter.
10
=> "a"

15
=> "f"

# Digits on both sides of the step from 9 to a: 9 * 16 + 10.
154
=> "9a"

# Zeros inside the number and at its end stay: 1 * 16^4 + 2 * 16^2 + 3.
66051
=> "10203"

# Every digit a letter.
11259375
=> "abcdef"

# 16^7, the smallest number with eight digits, and the number before it.
268435456
=> "10000000"

268435455
=> "fffffff"

# The largest int.
2147483647
=> "7fffffff"

# The smallest int, -2^31; negating it overflows an int.
-2147483648
=> "80000000"

# -1 becomes 2^32 - 1, the largest value any num is written as: every
# digit is f.
-1
=> "ffffffff"

# -2^31 + 1 leaves zeros inside a negative number's digits.
-2147483647
=> "80000001"
