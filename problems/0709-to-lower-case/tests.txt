# To Lower Case.

@example R, B, F and C become small letters; the digits, spaces and comma stay.
"Room 12B, Floor C"
=> "room 12b, floor c"

@example The brackets, the _ and the @ are close to the capitals in ASCII, but they are not letters, so they stay.
"[MAX_SIZE]@Q"
=> "[max_size]@q"

@example Small letters stay small; only the capitals change.
"sOME wORDS"
=> "some words"

# No capital at all, so nothing changes.
"already quiet."
=> "already quiet."

# Every letter is a capital.
"XYZZY"
=> "xyzzy"

# Single characters: a capital, a small letter, a non-letter.
"A"
=> "a"

"z"
=> "z"

"~"
=> "~"

# The characters on each side of both alphabets in ASCII.
"@AZ[`az{"
=> "@az[`az{"

# The six characters that c | 32 would change though they aren't capitals.
"@[\\]^_"
=> "@[\\]^_"

# Small letters must stay small (flipping case would turn them into capitals).
"MiXeD cAsE"
=> "mixed case"

# Digits, the space and punctuation.
"0123456789 !#$%&'()*+,-./:;<=>?{|}~"
=> "0123456789 !#$%&'()*+,-./:;<=>?{|}~"

# A capital at the first and at the last index.
"Quiz Q"
=> "quiz q"

# Spaces at both ends, and nothing but spaces.
"  Hi THERE  "
=> "  hi there  "

"   "
=> "   "

# Quotes and backslashes.
"Say \"OK\" \\ BYE"
=> "say \"ok\" \\ bye"

# The whole alphabet of capitals.
"ABCDEFGHIJKLMNOPQRSTUVWXYZ"
=> "abcdefghijklmnopqrstuvwxyz"

# The longest input, 100 characters ending in a capital.
"Bring 3 CRATES of [PLUMS] and `FIGS` to Dock_7 @ NOON; tell Mr. Q {URGENT} ~ thanks, meet at GATE QZ"
=> "bring 3 crates of [plums] and `figs` to dock_7 @ noon; tell mr. q {urgent} ~ thanks, meet at gate qz"
