# Reverse String Prefix.

@example The prefix "plan" reversed is "nalp"; the "et" after it stays in place.
"planet"
4
=> "nalpet"

@example k is the whole length, so all of s is reversed.
"stone"
5
=> "enots"

@example The prefix "racecar" reads the same backwards, so nothing changes.
"racecars"
7
=> "racecars"

# A single letter.
"m"
1
=> "m"

# k = 1: a prefix of one letter, so s comes back as it was.
"jump"
1
=> "jump"

# k = 2 swaps the first two letters.
"oxen"
2
=> "xoen"

# k = length - 1: only the last letter stays put.
"trumpet"
6
=> "epmurtt"

# Even k: every letter of the prefix moves.
"wizard"
4
=> "aziwrd"

# Odd k: the middle letter of the prefix stays where it is.
"monkey"
5
=> "eknomy"

# The letter right after the prefix differs from the prefix's last letter,
# so reversing one letter too many or too few shows.
"lmnop"
3
=> "nmlop"

# Repeated letters: a prefix that reads the same backwards, and one that
# does not.
"abbaxyz"
4
=> "abbaxyz"

"aaab"
4
=> "baaa"

# The longest strings: 100 letters, all of them reversed, then all but the
# last.
"lanternsglowoverquietharborsatduskwhilefishermenmendtheirnetsandgullscircleaboveslowtidesdriftpastus"
100
=> "sutsaptfirdseditwolsevobaelcricsllugdnastenriehtdnemnemrehsifelihwksudtasrobrahteiuqrevowolgsnretnal"

"lanternsglowoverquietharborsatduskwhilefishermenmendtheirnetsandgullscircleaboveslowtidesdriftpastus"
99
=> "utsaptfirdseditwolsevobaelcricsllugdnastenriehtdnemnemrehsifelihwksudtasrobrahteiuqrevowolgsnretnals"
