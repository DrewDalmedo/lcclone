# Destination City.

@example The legs chain Hobart, Perth, Darwin and Cairns, and no leg leaves Cairns.
[["Perth","Darwin"],["Hobart","Perth"],["Darwin","Cairns"]]
=> "Cairns"

@example A single leg finishes where it drives to.
[["Hilo","Kona"]]
=> "Kona"

@example Ann, Port Ann and port ann are three different cities: the trip goes from Ann to Port Ann, then on to port ann.
[["Port Ann","port ann"],["Ann","Port Ann"]]
=> "port ann"

# The leg into the last city is listed first. The first city, Al, is also
# named only once.
[["Cy","Dee"],["Bo","Cy"],["Al","Bo"]]
=> "Dee"

# The leg into the last city is listed in the middle.
[["Bb","Cc"],["Dd","Ee"],["Aa","Bb"],["Cc","Dd"]]
=> "Ee"

# The last city's name is the start of names that set out.
[["Santana","Santa"],["Santa","San"],["Santo","Santana"]]
=> "San"

# The last city's name continues names that set out.
[["Sant","Santa"],["Santa","Santana"],["San","Sant"]]
=> "Santana"

# Cities whose names differ only in case: legs set out from the last
# city's name in lower case and in capitals.
[["RIO","rio"],["rio","Rio"],["rIo","RIO"]]
=> "Rio"

# Names of two words that share a word: the last city's first word is
# also the first word of a city that a leg sets out from.
[["Las Vegas","El Paso"],["El Paso","Las Cruces"],["Santa Fe","Las Vegas"]]
=> "Las Cruces"

# A doubled space and a trailing space make other names.
[["New Bay ","New  Bay"],["New  Bay","New Bay"]]
=> "New Bay"

# The same name with no space, a trailing space and a doubled space: all
# four are different cities, and only "Old Mill" has no leg out.
[["Old Mill ","Old  Mill"],["Old  Mill","Old Mill"],["OldMill","Old Mill "]]
=> "Old Mill"

# Names of 10 characters, the longest allowed.
[["Copenhagen","Manchester"],["Valparaiso","Copenhagen"]]
=> "Manchester"

# Anagrams: a hash that only adds up the letters can't tell them apart.
[["abc","cab"],["cab","acb"],["bca","abc"]]
=> "acb"
