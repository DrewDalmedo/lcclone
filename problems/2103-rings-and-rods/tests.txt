# Rings and Rods.

@example Rod 7 gets a red, a blue and a green ring; rod 2 only a green and a red one.
"R7G2B7G7R2"
=> 1

@example Six rings on rod 5, but none of them is red.
"B5B5B5G5G5G5"
=> 0

@example Rods 0 and 9, at the two ends of the row, both get one ring of each colour.
"G0B9R0B0R9G9"
=> 2

# A single ring.
"B8"
=> 0

# Two colours on one rod.
"R3G3"
=> 0

# The fewest rings that complete a rod: blue, green, red.
"B4G4R4"
=> 1

# A rod completed twice over still counts once.
"R3G3B3R3G3B3"
=> 1

# Three rings on rod 4 in two colours, the third a repeat of the first:
# counting colour changes instead of colours would call it complete.
"R4G4R4"
=> 0

# Each rod lacks a different colour.
"R1G1G2B2B3R3"
=> 0

# The very last ring completes rod 6.
"G6R6R6G6B1B6"
=> 1

# Three rods, their rings interleaved, the colours in a different order
# on each.
"B1G2R3G1R2B3R1B2G3"
=> 3

# All ten rods complete.
"R0G1B2R3G4B5R6G7B8R9G0B1R2G3B4R5G6B7R8G9B0R1G2B3R4G5B6R7G8B9"
=> 10

# 100 red rings, ten on each rod: nothing is complete.
"R0R1R2R3R4R5R6R7R8R9R0R1R2R3R4R5R6R7R8R9R0R1R2R3R4R5R6R7R8R9R0R1R2R3R4R5R6R7R8R9R0R1R2R3R4R5R6R7R8R9R0R1R2R3R4R5R6R7R8R9R0R1R2R3R4R5R6R7R8R9R0R1R2R3R4R5R6R7R8R9R0R1R2R3R4R5R6R7R8R9R0R1R2R3R4R5R6R7R8R9"
=> 0

# 100 rings: rods 0 to 8 are complete, and rod 9 has red and green rings
# but no blue one.
"G2G2R4G3R7R0G8R2B8B1B5B7R5R3G7B8B2B4R7R2B8R6R3G7B1G6G2R8G8B3B6R9R8B2R0B6G1R3B0B3G0R3R4R3B7R2G6G3B4G0G9B0R6G4B7G3G2G8R1R3R9R0G5R7B1R2R4B6R9R6G7G9B4B0B4G0G6G4B0G9B6G0R0G2R4G2B8G3G7R0G3G1G3G2R5R2R1R7G8B0"
=> 9
