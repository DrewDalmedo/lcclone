# String Matching in an Array. compare: unordered 0; the expected answers
# list the words in the order they appear in words.

@example tor and or hide in storm, and rest in forest. or also hides in tor and forest, but is listed once.
["storm","tor","or","rest","forest"]
=> ["tor","or","rest"]

@example lan is a run of planet; pnt takes letters of planet that aren't next to each other.
["planet","pnt","lan"]
=> ["lan"]

@example No word appears inside another.
["sun","moon","star"]
=> []

# A single word.
["a"]
=> []

# A chain: a hides in ab, and ab in abc.
["abc","ab","a"]
=> ["ab","a"]

# The word comes after the only word it hides in.
["hello","ell"]
=> ["ell"]

# Hiding at the start, in the middle and at the end of a word.
["keyboard","key","boa","ard","board"]
=> ["key","boa","ard","board"]

# Overlapping occurrences.
["ababa","aba","bab"]
=> ["aba","bab"]

# A matcher that skips ahead after a partial match misses this one.
["aaab","aab"]
=> ["aab"]

# A partial match that runs off the end of the longer word.
["abc","bcd"]
=> []

# Words of the same length never hide in each other.
["abc","bca","cab"]
=> []

# Words of one letter.
["q","w","qw","e"]
=> ["q","w"]

# Hiding in two words, one before it and one after it: listed once.
["mnop","no","pqno"]
=> ["no"]

# The same letters in reverse order don't count.
["stop","pots","top"]
=> ["top"]

# The longest words: 29 letters hiding in 30.
["abcdefghijklmnopqrstuvwxyzabcd","zzzzzzzzzzzzzzzzzzzzzzzzzzzzzz","bcdefghijklmnopqrstuvwxyzabcd"]
=> ["bcdefghijklmnopqrstuvwxyzabcd"]
