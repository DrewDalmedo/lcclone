# Final Value of Variable After Performing Operations.

@example One statement adds 1 and three take 1 away: 1 - 3 = -2.
["X--","++X","X--","X--"]
=> -2

@example Five steps up and one step down.
["X++","X++","--X","X++","++X","X++"]
=> 4

@example Three steps up and three steps down cancel out.
["X--","X++","X--","++X","++X","--X"]
=> 0

# One statement of each form.
["++X"]
=> 1

["X++"]
=> 1

["--X"]
=> -1

["X--"]
=> -1

# Only postfix increments: the first character is X, yet X goes up.
["X++","X++","X++"]
=> 3

# Only prefix decrements: the last character is X, yet X goes down.
["--X","--X"]
=> -2

# All four forms, more steps down than up: 2 - 5.
["X++","X--","X--","--X","++X","X--","--X"]
=> -3
