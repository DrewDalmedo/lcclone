# Best Poker Hand.

@example All five cards have suit c.
[6,11,2,8,4]
["c","c","c","c","c"]
=> "Flush"

@example Three 8s count, even with a pair of 2s beside them.
[8,2,8,2,8]
["a","b","c","d","d"]
=> "Three of a Kind"

@example The ranks 4 to 8 are in a row, but no rank repeats and the suits differ.
[7,5,4,8,6]
["b","d","b","a","c"]
=> "High Card"

# Four cards of one rank.
[9,3,9,9,9]
["c","c","a","d","b"]
=> "Three of a Kind"

# Exactly three of one rank.
[12,4,12,6,12]
["d","d","b","b","a"]
=> "Three of a Kind"

# Three of one rank and two of another, the pair first.
[6,6,11,11,11]
["a","b","a","c","d"]
=> "Three of a Kind"

# Four cards of rank 13 and one of rank 1.
[13,13,13,1,13]
["d","c","b","d","a"]
=> "Three of a Kind"

# Two pairs.
[3,10,10,3,5]
["a","b","c","d","a"]
=> "Pair"

# One pair, of the highest rank, beside rank 1.
[13,1,13,7,4]
["b","b","c","d","a"]
=> "Pair"

# The pair is the last two cards.
[2,5,8,11,11]
["a","b","c","d","a"]
=> "Pair"

# Ranks in a row, all one suit: just a flush.
[9,10,11,12,13]
["d","d","d","d","d"]
=> "Flush"

# A flush in suit b.
[1,3,5,7,13]
["b","b","b","b","b"]
=> "Flush"

# Four cards of one suit and one of another are not a flush.
[2,6,9,11,13]
["b","b","b","b","c"]
=> "High Card"

[4,4,8,10,12]
["c","a","a","a","a"]
=> "Pair"

# 10 to 13 and then 1, one card off the suit: nothing special.
[10,11,12,13,1]
["c","c","d","c","c"]
=> "High Card"

# Random small cases from gen.c (lc check --regen makes them).
@gen 1 small
[7,8,10,7,9]
["c","c","c","d","d"]
=> "Pair"

@gen 2 small
[6,4,5,3,2]
["a","a","a","a","a"]
=> "Flush"

@gen 3 small
[7,6,6,11,12]
["a","c","b","d","a"]
=> "Pair"

@gen 4 small
[8,11,4,5,13]
["a","d","d","a","b"]
=> "High Card"

@gen 5 small
[2,4,4,3,2]
["d","d","a","c","c"]
=> "Pair"

@gen 6 small
[3,2,5,3,5]
["c","c","a","b","b"]
=> "Pair"

@gen 7 small
[8,8,9,10,8]
["b","a","a","d","d"]
=> "Three of a Kind"

@gen 8 small
[3,3,4,4,3]
["c","d","b","d","b"]
=> "Three of a Kind"

# Full-size random cases (made the same way: a hand always has five cards).
@gen 9
[4,5,3,2,3]
["d","c","d","c","c"]
=> "Pair"

@gen 10
[2,10,1,11,7]
["a","b","b","c","a"]
=> "High Card"
