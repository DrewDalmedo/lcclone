# Detect Capital.

@example The first letter is a capital and no other letter is.
"Marble"
=> true

@example Every letter is a capital.
"QUIZ"
=> true

@example The first letter is small, so the K would have to be small too.
"toKen"
=> false

# A single letter, in either case.
"q"
=> true

"W"
=> true

# Two letters: a capital after a small first letter is the one bad shape.
"Ab"
=> true

"aB"
=> false

# The only capital is the last letter: one capital is fine only at the start.
"rivetS"
=> false

# Two capitals first, then a small letter, deep inside or at the very end.
"CANyON"
=> false

"PLANKs"
=> false

# Two capitals, then small letters only: index 1 is the odd one out.
"MArsh"
=> false

# A capital, a small letter, then capitals: index 1 is the odd one out.
"GrOUND"
=> false

# A capital and a small letter first, then a capital deep inside or at the end.
"SiLver"
=> false

"ButtoN"
=> false

# The longest words, of 100 letters: capitalised, and small but for the last.
"Mlstbjpecgiqzdynkarvhowxfumlstbjpecgiqzdynkarvhowxfumlstbjpecgiqzdynkarvhowxfumlstbjpecgiqzdynkarvho"
=> true

"mlstbjpecgiqzdynkarvhowxfumlstbjpecgiqzdynkarvhowxfumlstbjpecgiqzdynkarvhowxfumlstbjpecgiqzdynkarvhO"
=> false
