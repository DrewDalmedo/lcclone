# The smallest range: 1 alone, whose one 1 is not a prime count.
1
1

# Only powers of two.
1
2
=> 0

# Two 1s: 2 is prime.
3
3
=> 1

# Both ends win and the number between them doesn't: 31 = 11111 and
# 33 = 100001. Testing x itself for primality would count 31 but not 33.
31
33
=> 2

# 65535 has sixteen 1s, 65536 one and 65537 = 2^16 + 1 two. A count that
# only looks at the low 16 bits, or a 16-bit x, misses 65537.
65535
65537
=> 1

# Twelve 1s (8190, not prime), thirteen (8191, prime), then 8192 = 2^13.
8190
8192
=> 1

# 262142 = 2^18 - 2 has seventeen 1s, a prime, and 262143 = 2^18 - 1 has
# eighteen, which isn't. A list of primes that stops at 13 or 16 misses
# 17.
262142
262144
=> 1

# Nineteen 1s, the most a number up to 10^6 can have: 983039 = 2^20 - 1 -
# 2^16. A table of primality by count needs 20 entries.
983039
983039
=> 1
