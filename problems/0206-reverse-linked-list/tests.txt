# Reverse Linked List. Lists are written from head to tail.

@example The last node, 1, is now first.
[8,3,6,1]
=> [1,6,3,8]

@example Negative values move like any other.
[-4,10,2]
=> [2,10,-4]

@example One node stays where it is.
[7]
=> [7]

# The empty list.
[]
=> []

# The value limits.
[-5000,5000]
=> [5000,-5000]

# All equal.
[5,5,5]
=> [5,5,5]

# A palindrome.
[4,2,4]
=> [4,2,4]

# Odd length.
[10,20,30]
=> [30,20,10]

# Descending negatives.
[0,-1,-2,-3,-4]
=> [-4,-3,-2,-1,0]

# Repeats in the middle.
[9,1,1,2]
=> [2,1,1,9]

@gen 1 small
[-2,-3,2,0,3,3]
=> [3,3,0,2,-3,-2]

@gen 2 small
[3,-3,-2,-3]
=> [-3,-2,-3,3]

@gen 3 small
[0,2,3,-3,2,1,-3]
=> [-3,1,2,-3,3,2,0]

@gen 4 small
[-3,3,-3]
=> [-3,3,-3]

@gen 5 small
[3,-2,2,2,0]
=> [0,2,2,-2,3]
