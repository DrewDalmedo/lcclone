# Unique Middle Element.

@example The centre holds 8, found nowhere else; 6 and 2 repeat, but neither is at the centre.
[6,2,8,6,2]
=> true

@example The centre (position 3) holds 5, and so does position 5.
[9,4,1,5,9,5,3]
=> false

@example A single entry is the centre, and nothing else can share its value.
[42]
=> true

# The smallest input, at both ends of the value range.
[1]
=> true

[100]
=> true

# All entries equal.
[5,5,5]
=> false

# 100 at the centre, with a copy (a count table of 100 slots indexed by
# value would overflow), and 100 repeated around a lone centre.
[100,100,1]
=> false

[100,1,100]
=> true

# The only copy of the centre's value is far away: at the very start, or
# at the very end. Its neighbours differ from it.
[4,1,2,3,4,5,6,8,9]
=> false

[3,8,6,1,2,9,1]
=> false

# The entry just after the centre has a copy, but the centre doesn't.
[2,3,5,4,4]
=> true

# The other way round: the centre has a copy, the entry after it doesn't.
[3,8,7,1,7]
=> false

# Sorting first would put another value at the centre: 3 (alone) here...
[1,9,9,2,3]
=> false

# ... and 4 (repeated) here.
[4,4,1,6,7]
=> true

# Every position but one holds the same value: the centre's, or not.
[8,8,8,8,8,8,3]
=> false

[8,8,8,1,8,8,8]
=> true
