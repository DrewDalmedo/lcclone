# Single Number.

@example The two 7s form a pair, and -3 is the extra element.
[7,-3,7]
=> -3

@example The pairs are the 20s and the 9s, neither side by side; 6 is the extra element.
[20,6,9,20,9]
=> 6

@example -4 and 4 are different values: the two -4s form a pair, and 4 is the extra element.
[-4,4,-4]
=> 4

# One element, at the upper bound and 0.
[30000]
=> 30000

[0]
=> 0

# The lone value is 0.
[3,0,3]
=> 0

# The lone value comes first, and the pairs are mirrored after it.
[-15,8,30,30,8]
=> -15

# The lone value comes last: comparing neighbours after sorting must not
# read past the end.
[6,13,6,13,27]
=> 27

[-4,-4,0,0,11,11,19]
=> 19

# Sorted, with the lone value in the middle.
[-2,-2,5,7,7]
=> 5

# The copies as far apart as they can be, around the lone value.
[1,2,3,4,5,4,3,2,1]
=> 5

# A value and its negation, each as a pair of its own.
[100,-100,-100,55,100]
=> 55

[-9,9,-1,9,-9]
=> -1

# The extreme values.
[30000,-30000,30000]
=> -30000

[-30000,30000,-30000,29999,29999]
=> 30000

# Random small cases from gen.c (lc check --regen makes them).
@gen 1 small
[-3,0,3,3,1,1,-3]
=> 0

@gen 2 small
[0,-3,0]
=> -3

@gen 3 small
[-5]
=> -5

@gen 4 small
[-4,-3,-4,2,2,-3,0]
=> 0

@gen 5 small
[-2,-1,2,-3,-2,3,2,-3,3]
=> -1

@gen 6 small
[-2,-2,3]
=> 3

@gen 7 small
[-1,-1,3]
=> 3

@gen 8 small
[0,0,1]
=> 1
