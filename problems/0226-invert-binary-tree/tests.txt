# Invert Binary Tree. A case is the tree in level order; the result is the
# turned tree in level order.

@example The root's children swap sides, and so do the lone children 8 and 6 below them.
[12,5,-3,null,8,6]
=> [12,-3,5,null,6,8]

@example A lone left child becomes a lone right child.
[3,7]
=> [3,null,7]

@example A path of right children becomes a path of left children.
[1,null,2,null,-4]
=> [1,2,null,-4]

# An empty tree.
[]
=> []

# One node, at the lowest value.
[-100]
=> [-100]

# A lone left child, at the highest value.
[5,100]
=> [5,null,100]

# A path of left children turns into a path of right children.
[1,2,null,3,null,4]
=> [1,null,2,null,3,null,4]

# A path of right children turns into a path of left children.
[9,null,8,null,7]
=> [9,8,null,7]

# A zigzag flips at every turn.
[1,2,null,null,3,4]
=> [1,null,2,3,null,null,4]

# A complete tree with one node on its fourth level: every level reverses.
[1,2,3,4,5,6,7,8]
=> [1,3,2,7,6,5,4,null,null,null,null,null,null,null,8]

# Asymmetric subtrees three levels deep: inverting one side twice is caught.
[1,2,3,4,null,null,5,6]
=> [1,3,2,5,null,null,4,null,null,null,6]

# Equal values: only the shape shows the swap.
[2,2,2,2]
=> [2,2,2,null,null,null,2]

# Zero and negatives, with a one-child node.
[0,-1,1,null,-2]
=> [0,1,-1,null,null,-2]

# One-child nodes at several depths.
[5,1,null,2,3,null,4]
=> [5,null,1,3,2,null,null,4]
