# Thousand Separator.

@example Four digits: a dot goes between the 3rd and the 4th digit from the right, and the zeros stay.
7006
=> "7.006"

@example Six digits make two full groups, with no dot in front of the first.
314159
=> "314.159"

@example Two digits get no dot.
40
=> "40"

# Zero is the single digit 0.
0
=> "0"

# The largest number with no dot, and the smallest with one.
999
=> "999"

1000
=> "1.000"

# Nine digits: three full groups, still no dot in front.
100200300
=> "100.200.300"

999999999
=> "999.999.999"

# Zeros at the start and at the end of a group stay.
12005
=> "12.005"

1000000
=> "1.000.000"

1040000
=> "1.040.000"

# Ten digits, the most an int has: a power of ten kept in an int overflows
# when it is multiplied past 10^9.
1000000000
=> "1.000.000.000"

2000000001
=> "2.000.000.001"

2147483647
=> "2.147.483.647"
