# Valid Perfect Square.

@example 169 tiles make a square of 13 rows of 13.
169
=> true

@example 200 lies between 14 * 14 = 196 and 15 * 15 = 225.
200
=> false

@example A single tile is a square of one row of one.
1
=> true

# The two smallest numbers that aren't squares, then 2 * 2.
2
=> false

3
=> false

4
=> true

1073741824
=> true

# 46339 * 46339 and 46340 * 46340, the two largest squares that fit in a
# 32-bit int. Squaring any guess above 46340 in 32 bits overflows.
2147302921
=> true

2147395600
=> true

2147395599

=> false
