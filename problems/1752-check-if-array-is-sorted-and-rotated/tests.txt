# Check if Array Is Sorted and Rotated.

@example Cut [1,4,6,6,9,9] after its third value and put the back piece [6,9,9] first.
[6,9,9,1,4,6]
=> true

@example From [2,3,7,8] the cuts give [2,3,7,8], [3,7,8,2], [7,8,2,3] and [8,2,3,7], never this.
[2,7,3,8]
=> false

@example Already sorted: the cut before the first value changes nothing.
[3,3,8,10]
=> true

# A single value.
[42]
=> true

# Two values going down: [1,100] cut in the middle.
[100,1]
=> true

# Three values going down.
[3,2,1]
=> false

# All equal: the values never go down at all.
[7,7,7,7]
=> true

# One step down inside, but the last value is larger than the first.
[4,5,1,2,6]
=> false

# The smallest value at both ends: the cut is not before its first copy.
[1,5,8,1,1]
=> true

# A run of equal values split between the end and the start.
[5,5,8,3,5]
=> true

# The largest value at both ends: starting right after its last copy
# doesn't give the sorted order.
[9,2,4,6,9]
=> true

# Two steps down inside, though each piece is sorted.
[4,6,2,3,1]
=> false

# The step down right after the first value, or right before the last.
[9,1,2,3]
=> true

[2,3,4,1]
=> true

# A sorted array with two of its values swapped: the same values, but no
# cut gives this order.
[1,6,3,4,2]
=> false

# 100 values with repeats; the cut falls inside the run of 64s.
[64,65,65,68,68,68,69,69,71,72,74,75,76,78,78,78,79,79,79,80,83,83,83,85,86,87,91,91,91,91,92,94,95,96,97,99,100,1,2,2,2,4,4,4,5,5,6,7,7,9,10,10,11,12,14,15,15,17,18,19,19,20,20,21,23,23,24,28,29,32,32,32,33,35,35,35,36,36,37,37,40,41,41,42,42,44,45,47,48,51,52,52,54,55,55,55,55,59,62,64]
