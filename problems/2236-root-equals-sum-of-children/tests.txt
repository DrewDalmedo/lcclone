# Root Equals Sum of Children. A case is the three nodes in level order: root,
# left child, right child; the result says whether the root is their sum.

@example 9 + (-2) = 7, the claimed total.
[7,9,-2]
=> true

@example The items add up to -3, one above the total.
[-4,-1,-2]
=> false

# All zero.
[0,0,0]
=> true

# One item is the whole total.
[25,0,25]
=> true

# Items that cancel to zero.
[0,100,-100]
=> true

# The sum is one above the total.
[12,7,6]
=> false

# The sum is one below the total.
[12,7,4]
=> false

# The absolute values add up, but the signed values don't.
[-5,2,3]
=> false

# The signed values add up, but the absolute values don't.
[1,-4,5]
=> true

# The difference of the items is the total; their sum isn't.
[3,5,2]
=> false

# Both items at 100: the sum is past any total.
[100,100,100]
=> false

# Both items at -100.
[-100,-100,-100]
=> false

# The total at its lower limit, reached exactly.
[-100,-37,-63]
=> true

# 100 + 100 would wrap to -56 in 8 bits.
[-56,100,100]
=> false

@gen 1 small
[-3,-2,2]
=> false

@gen 2 small
[1,3,-2]
=> true

@gen 3 small
[-2,1,-3]
=> true

@gen 6 small
[5,-1,-4]
=> false

@gen 10 small
[-7,-3,2]
=> false

@gen 13 small
[-1,1,-2]
=> true

@gen 14 small
[8,4,4]
=> true

@gen 19 small
[-4,3,1]
=> false
