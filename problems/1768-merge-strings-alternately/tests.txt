# Merge Strings Alternately.

@example Turns hand over h, e, l, l; word2 is then used up, so the o still held by word1 goes last.
"hlo"
"el"
=> "hello"

@example Turns hand over c, w, a, x, t; word1 is then used up, so y and z follow together.
"cat"
"wxyz"
=> "cwaxtyz"

@example word1 takes the first turn although it is shorter. After m it has nothing left, so all of north follows.
"m"
"north"
=> "mnorth"

# Both words one letter long; word1 still goes first, though its letter
# comes later in the alphabet.
"b"
"a"
=> "ba"

# The same length: a perfect weave with nothing left over.
"acegi"
"bdfhj"
=> "abcdefghij"

# word1 much longer: all of it after the first letter is left over.
"abcdefgh"
"z"
=> "azbcdefgh"

# word2 much longer.
"q"
"rstuvw"
=> "qrstuvw"

# The shorter word, and the one later in the alphabet, is word1: it leads.
"zy"
"abcd"
=> "zaybcd"

# Identical words.
"xyz"
"xyz"
=> "xxyyzz"

# One letter repeated: any order of turns looks the same here.
"aaa"
"aaaaa"
=> "aaaaaaaa"

# Longest word1, one-letter word2.
"abcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghij"
"z"
=> "azbcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghij"

# One-letter word1, longest word2.
"z"
"abcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghij"
=> "zabcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghijabcdefghij"

# Both words at the limit of 100 letters: 200 letters in the result.
"abababababababababababababababababababababababababababababababababababababababababababababababababab"
"cdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcdcd"
