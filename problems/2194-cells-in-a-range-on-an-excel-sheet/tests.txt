# Cells in a Range on an Excel Sheet.

@example Columns C and D, rows 3 to 5: the three cells of column C come first, then those of column D.
"C3:D5"
=> ["C3","C4","C5","D3","D4","D5"]

@example Both corners are the same cell, so the block is that one cell.
"M7:M7"
=> ["M7"]

@example A block one row tall, from column W to the last column.
"W2:Z2"
=> ["W2","X2","Y2","Z2"]

# The first cell of the sheet on its own.
"A1:A1"
=> ["A1"]

# The last cell of the sheet on its own.
"Z9:Z9"
=> ["Z9"]

# One column, every row.
"B1:B9"
=> ["B1","B2","B3","B4","B5","B6","B7","B8","B9"]

# One row, every column.
"A5:Z5"

# Two by two in the bottom-right corner: Y8 Z8 Y9 Z9 would be row by row.
"Y8:Z9"
=> ["Y8","Y9","Z8","Z9"]

# Three columns of three rows.
"H4:J6"
=> ["H4","H5","H6","I4","I5","I6","J4","J5","J6"]

# Two columns, the bottom row only.
"P9:Q9"
=> ["P9","Q9"]
