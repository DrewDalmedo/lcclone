# Goat Latin.

@example Old and eat start with vowels and stay; rams moves its r to the back. The bleats are maa, maaa and maaaa.
"Old rams eat"
=> "Oldmaa amsrmaaa eatmaaaa"

@example Y is a consonant, so Yak moves its Y to the back, as My moves its M; both stay capitals.
"My Yak"
=> "yMmaa akYmaaa"

@example Moving the only letter of a word leaves it as it was.
"x"
=> "xmaa"

# A single word that starts with a capital vowel.

=> "Emaa"

# Every vowel, in both cases, starting a word: no word changes but for its bleat.

=> "inkmaa Apemaaa urnmaaaa Orbmaaaaa elkmaaaaaa Innmaaaaaaa axemaaaaaaaa Evemaaaaaaaaa oakmaaaaaaaaaa Ughmaaaaaaaaaaa"

# Capital consonants move to the back and stay capitals.
"Zebra Kiwi"
=> "ebraZmaa iwiKmaaa"

# y and Y are consonants.
"yes Yo"
=> "esymaa oYmaaa"

# One-letter words, a vowel and a consonant.
"I b"
=> "Imaa bmaaa"

# The same word three times gets a longer bleat each time.
"go go go"
=> "ogmaa ogmaaa ogmaaaa"

# A consonant first, then nothing but vowels: only the first letter decides.
"queue"
=> "ueueqmaa"

# Letters after the first keep their case.
"sKY hIGH"
=> "KYsmaa IGHhmaaa"

=> "qmaa Rmaaa smaaaa Tmaaaaa vmaaaaaa Wmaaaaaaa xmaaaaaaaa Zmaaaaaaaaa bmaaaaaaaaaa Cmaaaaaaaaaaa"
