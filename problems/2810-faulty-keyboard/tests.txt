# Faulty Keyboard.

@example After m, o, s and a the line is "mosa"; the i turns it into "asom", and c goes on the end.
"mosaic"
=> "asomc"

@example Turning the one-letter line "k" changes nothing; the last i turns "kw" into "wk".
"kiwi"
=> "wk"

@example Three turns in a row leave "de" as "ed", just as one turn would; then f is added.
"deiiif"
=> "edf"

# A single key.
"q"
=> "q"

# No i at all: the line is s itself.
"keyboard"
=> "keyboard"

# Two turns in a row cancel out.
"abiic"
=> "abc"

# A turn in the middle only turns the letters before it: removing the i's
# and reversing once, since there is one i, would give "dcba".
"abicd"
=> "bacd"

# Two turns, so an even count, but at different places: they don't cancel.
"abicdie"
=> "dcabe"

# A turn after the last letter turns the whole line.
"abci"
=> "cba"

# An even run of turns at the end leaves the line as it was.
"abcii"
=> "abc"

# Every other key is an i, ending with one.
"aibicidi"
=> "dbac"

# The longest s: one letter, then 99 presses of i. Turning a single letter
# changes nothing.
"ziiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiii"
=> "z"

# 98 presses of i between two letters cancel out.
"xiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiiy"
=> "xy"

# The longest s: 99 letters, then one i that reverses them all.
"abcdefghjklmnopqrstuvwxyzabcdefghjklmnopqrstuvwxyzabcdefghjklmnopqrstuvwxyzabcdefghjklmnopqrstuvwxyi"
=> "yxwvutsrqponmlkjhgfedcbazyxwvutsrqponmlkjhgfedcbazyxwvutsrqponmlkjhgfedcbazyxwvutsrqponmlkjhgfedcba"

# 50 letters, each followed by an i: the even-numbered letters end up in
# front, last first, and the odd-numbered ones behind them, first first.
"aibicidieifigihijikiliminioipiqirisitiuiviwixiyiziaibicidieifigihijikiliminioipiqirisitiuiviwixiyizi"
=> "zxvtrpnljgecaywusqomkhfdbacegjlnprtvxzbdfhkmoqsuwy"
