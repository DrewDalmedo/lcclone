# Reverse Words in a String III.

@example Each word is spelled backwards where it is; the dot belongs to the last word, and the O keeps its case.
"Owls hunt at dusk."
=> "slwO tnuh ta .ksud"

@example A word of one character stays as it is.
"a bc def"
=> "a cb fed"

@example With no space, the whole string is a single word.
"Kayak"
=> "kayaK"

# A single character.
"q"
=> "q"

# A single word of two characters.
"Ab"
=> "bA"

# Only the last word changes, and only the first.
"x yz"
=> "x zy"

"pq r"
=> "qp r"

# Quotes, a colon and a backslash are part of their words (the line is
# "hi" said C:\dir, escaped in the literals).
"\"hi\" said C:\\dir"
=> "\"ih\" dias rid\\:C"

# The lowest and the highest printable characters after the space.
"!~ ~!"
=> "~! !~"

# Digits and punctuation.
"12,34 5.6"
=> "43,21 6.5"
