# Find Words Containing Character. compare: unordered 0; the expected
# answers list the indices in increasing order.

@example river, stone and leaf have an e; oak doesn't.
["river","stone","leaf","oak"]
"e"
=> [0,1,2]

@example banana has three a's, but its position is listed once.
["banana","kiwi","fig"]
"a"
=> [0]

@example Neither word has a z.
["cat","dog"]
"z"
=> []

# A single word that is just x.
["q"]
"q"
=> [0]

# A single word without x.
["a"]
"b"
=> []

# x is in every word.
["ax","xb","x","cxc"]
"x"
=> [0,1,2,3]

# x only as the first or the last letter of a word.
["kite","ask","oak","mile"]
"k"
=> [0,1,2]

# Repeated words give separate indices.
["moon","sun","moon","moon"]
"o"
=> [0,2,3]

# Only the last word has x.
["abc","def","ghz"]
"z"
=> [2]

# Only the first word has x.
["yes","no","ok"]
"y"
=> [0]

# The first letter of the alphabet...
["zebra","apple","fuzz","pizza"]
"a"
=> [0,1,3]

# ...and the last one.
["zebra","apple","fuzz","pizza"]
"z"
=> [0,2,3]

# The longest words: one is all x, the other has x only at the end.
["aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa","bbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbba","b"]
"a"
=> [0,1]
