# Backspace String Compare.

@example The backspace erases the mistyped x, so both logs leave "kitten".
"kitx#ten"
"kitten"
=> true

@example The two backspaces at the start of s find the box empty and do nothing, while in t the backspace erases w: both leave "up".
"##up"
"w#up"
=> true

@example s leaves "moe", since its backspace takes the p, but t leaves "mope".
"mop#e"
"mope"
=> false

# The shortest logs: one letter each.
"a"
"a"
=> true

# A lone backspace leaves the box empty; the other box holds a letter.
"#"
"z"
=> false

# Every letter is erased in both logs: two empty boxes match.
"abc###"
"q#"
=> true

# Backspaces in an empty box are not saved for the letters after them.
"###abc"
"abc"
=> true

"a##b"
"b"
=> true

# Runs of backspaces between letters: a, b, #, c, #, # leaves nothing
# before the d.
"ab#c##d"
"d"
=> true

# One text is a prefix of the other ("ab" and "abd"), then a suffix
# ("zbc" and "bc"), then a prefix the other way round ("inks" and "ink").
"ab"
"abc#d"
=> false

"zz#bc"
"bc"
=> false

"inks"
"inkx#"
=> false

# The same length and the same characters, but "a" against "b".
"ab#"
"a#b"
=> false

# The texts differ only in their last letter: "pqs" against "pqt".
"pqr#s"
"pqt"
=> false

# The same four letters with the same first and last letter, in another
# order: "trap" against "tarp".
"trax#p"
"taq#rp"
=> false

# The backspaces erase letters that differ between s and t.
"mox#on"
"moy#on"
=> true

# 130 backspaces in a row erase x, y and z and then find the box empty;
# a skip count kept in a signed char would wrap past 127.
"xyz##################################################################################################################################ok"
"ok"
=> true
