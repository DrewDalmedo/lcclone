# Redistribute Characters to Make All Strings Equal.

@example Two transfers, each taking a "k" from the first string and putting it in front of "oo", leave both strings as "kkoo".
["kkkkoo","oo"]
=> true

@example There is only one "t", but two equal strings would hold the same number of "t"s each, which needs an even total.
["tu","su"]
=> false

@example A single string is already the same as itself.
["zebra"]
=> true

# One string of 100 letters: true whatever the letters are.
["qabcdefghiabcdefghiabcdefghiabcdefghiabcdefghiabcdefghiabcdefghiabcdefghiabcdefghiabcdefghiabcdefghi"]
=> true

# Two one-letter strings that differ.
["a","b"]
=> false

# No letter is shared at the start, but each one can be split three ways.
["aaa","bbb","ccc"]
=> true

# One long string holds nearly everything; the others have one letter.
["xxyy","x","y"]
=> true

# Each letter occurs as often as the other, 4 times, which doesn't split
# three ways.
["aabb","ab","ab"]
=> false

# The letters occur 4 and 2 times: different counts, both even.
["ccccd","d"]
=> true

# Three strings and two different letters, each 3 times.
["ab","ab","ab"]
=> true

# Every count is a multiple of the first string's length, 2, but not of 3.
["aa","b","b"]
=> false

# The letters of the first string are fine; "z" and "y" occur once each.
["ab","abzy"]
=> false

# All 26 letters; "a" and "z" occur once, every other letter twice.
["abcdefghijklmnopqrstuvwxyz","bcdefghijklmnopqrstuvwxy"]
=> false

# All 26 letters, each twice except "z", the last letter, which occurs once.
["abcdefghijklmnopqrstuvwxyz","abcdefghijklmnopqrstuvwxy"]
=> false

# All 26 letters, each twice except "a", the first letter, which occurs once.
["bcdefghijklmnopqrstuvwxyz","abcdefghijklmnopqrstuvwxyz"]
=> false

# 258 copies of "a" split three ways (a count kept in a byte would wrap
# around to 2).
["aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa","aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa","aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa"]
=> true
