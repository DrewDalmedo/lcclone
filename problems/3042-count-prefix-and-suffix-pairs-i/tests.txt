# Count Prefix and Suffix Pairs I.

@example "on" bookends both copies of "onion", "o" bookends "oo", and the first "onion" bookends the second.
["on","onion","o","oo","onion"]
=> 4

@example "alfa" bookends "alfalfa" (the two copies share its fourth letter) and its own later copy; "alfalfa" is too long to bookend the last word.
["alfa","alfalfa","alfa"]
=> 2

@example "gone" only starts with "go" and "ago" only finishes with it; "gogo" does both.
["go","gone","ago","gogo"]
=> 1

# A single word has no pair.
["a"]
=> 0

# Two copies of a one-letter word.
["z","z"]
=> 1

# Equal words at the first and last positions; "k" and "kit" start "kite"
# but don't finish it.
["kite","k","kit","kite"]
=> 1

# A longer word before a shorter one never bookends it ...
["abcab","ab"]
=> 0

# ... while in the other order it does.
["ab","abcab"]
=> 1

# A word only in the middle, or only at the start, of a later word.
["b","abc","ab","abcabd"]
=> 0

# The last four letters of "abcdxabed" differ from "abcd" in an inner
# letter; "abcdabcd" really finishes with it.
["abcd","abcdxabed","abcdabcd"]
=> 1

# "abcdbacd" finishes with the letters of "abcd" in another order: it
# ends in "cd" but its last four letters start "ba".
["abcd","abcdbacd"]
=> 0

# "sees" and "sxxs" agree in their first and last letters only;
# "seesees" holds two copies of "sees" that share an s.
["sees","sxxs","seesees"]
=> 1

# Overlapping copies, up to ten letters long.
["aaa","aaaa","xyzx","xyzxyzxyzx"]
=> 2

# A one-letter word bookends the words that start and finish with it.
["q","qwq","qw","wq","q"]
=> 2

# Pairs are counted by position: three copies of "a" make three pairs, and
# each copy before "aa" bookends it.
["a","a","aa","a"]
=> 5
