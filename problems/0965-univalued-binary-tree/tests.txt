# Univalued Binary Tree. A case is the tree in level order, with null for a
# missing child.

@example All five nodes hold 4.
[4,4,4,null,4,4]
=> true

@example One leaf on the right holds 7, and the rest hold 0.
[0,0,0,0,null,null,7]
=> false

@example A single node is always one colour.
[37]
=> true

# All zeros.
[0,0,0]
=> true

# All zeros but the deepest node of a left path.
[0,0,null,0,null,1]
=> false

# All non-zero but one leaf that is 0.
[3,3,3,0]
=> false

# The top value everywhere.
[99,99,99]
=> true

# An inner node differs, and its child matches the root.
[6,9,6,6,null,null,6]
=> false

# Only the last node in level order differs.
[2,2,2,null,2,null,2,3]
=> false

# A right path whose far end differs.
[1,null,1,null,1,null,2]
=> false

# A left path of equal values.
[8,8,null,8,null,8]
=> true

# 4 and 6 average out to the root's 5.
[5,4,6]
=> false

# A left child of the right subtree differs.
[7,7,7,null,7,8]
=> false

# A zigzag of equal values.
[2,2,null,null,2,2]
=> true

# Only the last node in preorder differs.
[1,1,1,null,null,null,2]
=> false

# The two 3s cancel under XOR, leaving the root's 5.
[5,3,3]
=> false
