# Check Adjacent Digit Differences.

@example The biggest moves, 5 to 3 and 4 to 2, are falls of exactly 2.
"4565342"
=> true

@example The move from 5 to 9 is a rise of 4.
"13597"
=> false

@example Every move is a fall of 2; the first and last readings are 8 apart, but they are not neighbours.
"97531"
=> true

# A single move: no change, a rise of exactly 2, a fall of exactly 2.
"00"
=> true

"02"
=> true

"20"
=> true

# A move of exactly 3 is too much, up or down.
"03"
=> false

"30"
=> false

# The largest possible jump, both ways.
"90"
=> false

"09"
=> false

# The only bad move is a fall, in the middle; every rise is small.
"24685676"
=> false

# The only bad move is a rise, in the middle; every fall is small.
"65458876"
=> false

# Falls of 1 and 2 throughout are fine (a difference taken as unsigned
# would wrap around and reject them).
"97864231"
=> true

# Only the first move is bad.
"5122334"
=> false

# A leading zero is a reading like any other (read as a number, "0312"
# would lose it and look smooth).
"0312"
=> false

# The first and last readings are 9 apart but every move is 1; read as a
# loop, the last move would wrongly go from 9 back to 0.
"0123456789"
=> true

# 100 identical readings.
"7777777777777777777777777777777777777777777777777777777777777777777777777777777777777777777777777777"
=> true
