# Odd String Difference.

@example "moq", "ace" and "suw" all have the steps 2, 2, while "kmp" has 2, 3.
["moq","kmp","ace","suw"]
=> "kmp"

@example "teen", "yjjs" and "vggp" have the steps -15, 0, 9; "ohhm" has -7, 0, 5. The odd word may come first.
["ohhm","teen","yjjs","vggp"]
=> "ohhm"

@example "ed" and "lk" have the step -1, but "az" has the step 25, not -1: the alphabet does not wrap around.
["ed","az","lk"]
=> "az"

# Three words, the odd one first (a single step each).
["ca","ab","bc"]
=> "ca"

# The odd word second.
["xyz","abd","pqr"]
=> "abd"

# The odd word last, differing only in its last step.
["klm","cde","cdf"]
=> "cdf"

# Differing only in the first step.
["efg","qrs","prs","uvw"]
=> "prs"

# Steps 1, 12 against 11, 2: written as digits without a separator, both
# would read "112". The words also share their first and last letters.
["abn","aln","cdp"]
=> "aln"

# The odd word is an anagram of another word.
["bac","abc","cde","xyz"]
=> "bac"

# The odd word's steps are the others' in reverse order (3, 1 against 1, 3).
["abe","cdg","ade","pqt"]
=> "ade"

# The odd word's steps are the others' negated.
["cbcb","mnmn","bcbc","xyxy"]
=> "cbcb"

# Identical words make up the rest.
["zzz","zzz","zzy","zzz"]
=> "zzy"

# Twenty letters, spanning the whole alphabet; the odd word skips the l.
["ghijklmnopqrstuvwxyz","abcdefghijklmnopqrst","abcdefghijkmnopqrstu","bcdefghijklmnopqrstu"]
=> "abcdefghijkmnopqrstu"
