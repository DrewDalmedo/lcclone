# Special Array I.

@example Odd, even, odd, even, odd: every neighbouring pair is mismatched.
[7,10,3,8,5]
=> true

@example 14 and 20 stand side by side and are both even.
[8,3,14,20,1]
=> false

@example Two odd numbers next to each other.
[11,13]
=> false

# A single element, odd or even, has no pair to check.
[99]
=> true

[100]
=> true

# Two elements: both even, then even and odd.
[2,4]
=> false

[100,1]
=> true

# Starting with an even number.
[20,33,48,51,62,75]
=> true

# As many odd as even numbers, but in the wrong order.
[3,5,8,10]
=> false

# Equal neighbours always share their parity.
[17,8,8,17]
=> false

# Only the first pair is wrong.
[1,3,2,5,4,7]
=> false

# Only the last pair is wrong.
[2,9,4,11,6,13,15]
=> false
