# Two Sum IV - Input is a BST. A case is the tree in level order, then k; the
# result says whether two different nodes add up to k.

@example 7 and 18 add up to 25, and they lie on different sides of the root.
[12,-3,30,null,7,18]
25
=> true

@example 8 + 8 would make 16, but the tree has only one 8 and no other pair works.
[8,2,null,-5]
16
=> false

@example -20 and -11 make -31.
[-20,null,-4,-11]
-31
=> true

# One node can't pair with itself.
[6]
12
=> false

# k = 0 with a 0 in the tree and no opposite value.
[0,-3,8]
0
=> false

# k = 20000: only 10^4 used twice would reach it.
[10000,9998]
20000
=> false

# k = -20000: only -10^4 used twice would reach it.
[-10000,null,-9997]
-20000
=> false

# The largest pair of all.
[9999,100,10000]
19999
=> true

# The smallest pair of all.
[-9999,-10000,0]
-19999
=> true

# The smallest and largest values cancel out.
[0,-10000,10000]
0
=> true

# Both nodes deep, in different subtrees of the root.
[50,20,80,10,30,70,90,null,null,25]
95
=> true

# All values even and k odd: every pair is tried and none works.
[40,20,60,10,30,50,70]
75
=> false

# k = 10^5, with a complement far outside the range of values.
[10000,-10000]
100000
=> false

# k = -10^5, the same on the other side.
[-10000,null,10000]
-100000
=> false

# A negative total from values of mixed signs.
[3,-7,12,null,-2]
-9
=> true

# 10^4 is met before its partner in a walk from the root, and is still a
# complement that must be found.
[10000,-1]
9999
=> true

# k = 65547 reads as 11 in 16 bits, and 2 + 9 = 11, but nothing reaches 65547.
[6,2,9]
65547
=> false

# k = -65547 reads as -11 in 16 bits, and -9 + -2 = -11, but nothing reaches it.
[-6,-9,-2]
-65547
=> false

# One of the pair holds 0, which a table using 0 for "empty" can't see.
[4,0,11]
11
=> true
